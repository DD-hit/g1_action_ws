cmake_minimum_required(VERSION 3.16)
project(ele_realsense_pipeline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(realsense2 2.54.1 REQUIRED)
# The ROS underlay also contains a different librealsense SDK.
include_directories(BEFORE SYSTEM "${realsense2_INCLUDE_DIR}")
add_executable(ele_realsense_pipeline main.cpp)
ament_target_dependencies(ele_realsense_pipeline rclcpp sensor_msgs geometry_msgs tf2_ros)
target_link_libraries(ele_realsense_pipeline ${realsense2_LIBRARY} Eigen3::Eigen)
target_compile_options(ele_realsense_pipeline PRIVATE -Wall -Wextra -Wpedantic)
get_filename_component(ELE_SDK_PREFIX "${realsense2_INCLUDE_DIR}" DIRECTORY)
set(ELE_SDK_LIBRARY_DIR "${ELE_SDK_PREFIX}/lib")
set_target_properties(ele_realsense_pipeline PROPERTIES INSTALL_RPATH "${ELE_SDK_LIBRARY_DIR}")
install(TARGETS ele_realsense_pipeline DESTINATION bin)
