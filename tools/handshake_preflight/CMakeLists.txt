cmake_minimum_required(VERSION 3.16)
project(handshake_preflight LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_path(BRSD_IK_INCLUDE brsd_upbody/common/arm_ik_seeds.h HINTS /workspace/install/brsd_upbody_control/include REQUIRED)
add_executable(handshake_preflight main.cpp)
target_include_directories(handshake_preflight PRIVATE ${BRSD_IK_INCLUDE})
ament_target_dependencies(handshake_preflight rclcpp moveit_core moveit_ros_planning moveit_msgs)
target_link_libraries(handshake_preflight yaml-cpp)
# Optional subscriber-only helper; the planner target itself does not link SDK.
set(unitree_sdk2_DIR "/opt/unitree_sdk2/lib/cmake/unitree_sdk2")
find_package(unitree_sdk2 CONFIG REQUIRED)
add_executable(handshake_full_state_reader read_full_state.cpp)
target_link_libraries(handshake_full_state_reader unitree_sdk2 yaml-cpp)
